% 从提问到播报的前后端状态机
\begin{tikzpicture}[x=1mm,y=1mm]
  \path[use as bounding box] (0,0) rectangle (158,63);
  \tikzset{
    hhState/.style={
      draw=JinmenBlue, fill=PaleJade!52, line width=.72pt,
      rounded corners=1.2mm, minimum width=27mm, minimum height=14mm,
      text width=23mm, align=center, inner sep=1.2mm,
      font=\small\sffamily, text=Ink
    },
    hhStateGuard/.style={hhState,draw=Cinnabar,fill=RicePaper},
    hhStateFlow/.style={
      -{Latex[length=2mm,width=1.35mm]}, draw=JinmenBlue,
      line width=.72pt, line cap=round, line join=round
    },
    hhStateFail/.style={
      -{Latex[length=2mm,width=1.35mm]}, draw=Cinnabar,
      line width=.72pt, line cap=round, line join=round
    }
  }

  \node[hhState] (idle) at (14,48) {\textbf{待机}\\[-.2ex]\footnotesize 接收问题};
  \node[hhState] (retrieve) at (46,48) {\textbf{检索}\\[-.2ex]\footnotesize 返回候选证据};
  \node[hhState] (generate) at (78,48) {\textbf{生成}\\[-.2ex]\footnotesize 结构化事实段};
  \node[hhState] (verify) at (110,48) {\textbf{复核}\\[-.2ex]\footnotesize 引用・数字・语义};
  \node[hhState] (speech) at (144,48) {\textbf{播报}\\[-.2ex]\footnotesize 一次性令牌};

  \node[hhStateGuard,minimum width=42mm,text width=37mm] (refuse) at (55,11)
    {\textbf{固定拒答}\\[-.2ex]\footnotesize 越界、证据弱或校验失败};
  \node[hhStateGuard,minimum width=42mm,text width=37mm] (error) at (111,11)
    {\textbf{可见错误}\\[-.2ex]\footnotesize 云服务不可用，不伪造结果};

  \draw[hhStateFlow] (idle.east) -- (retrieve.west);
  \draw[hhStateFlow] (retrieve.east) -- (generate.west);
  \draw[hhStateFlow] (generate.east) -- (verify.west);
  \draw[hhStateFlow] (verify.east) -- (speech.west);
  % 所有异常线先落到框体下方的独立轨道，再向下分流；不穿越任何节点。
  \draw[draw=Cinnabar,line width=.72pt,line cap=round]
    (retrieve.south) -- (46,29)
    (generate.south) -- (78,29)
    (verify.south) -- (110,29)
    (speech.south) -- (144,29)
    (46,29) -- (144,29);
  \foreach \x in {46,78,110,144}{\fill[Cinnabar] (\x,29) circle (0.65mm);}
  \draw[hhStateFail] (55,29) -- (refuse.north);
  \draw[hhStateFail] (111,29) -- (error.north);
  \node[font=\scriptsize\sffamily,text=Cinnabar,fill=RicePaper,inner sep=.6mm]
    at (99,29) {异常分流};

  \node[font=\scriptsize\sffamily,text=BrickGrey,anchor=south]
    at (80,59) {SSE 仅用于传输当前服务端状态；新问题会取消旧请求};
